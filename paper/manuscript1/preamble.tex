% --- 基础宏包 ---
\usepackage[utf8]{inputenc}        % UTF-8 编码支持
\usepackage[T1]{fontenc}           % 字体编码
\usepackage{amsmath, amssymb, amsthm}  % 数学公式
\usepackage{graphicx}              % 图形插入
\usepackage{booktabs}              % 专业表格
\usepackage{array}                 % 表格增强

\usepackage{hyperref}              % 超链接

    urlcolor=linkcolor,
    pdfborder={0 0 0}
}
